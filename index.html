const SHEET_ID = '1xqACM_Ca1eLlp8EFaYTtJdbqLkE_nFivb52dhsIsoM0'; function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const title = payload.title || ''; const description = payload.description || ''; const fullText = (title ? title + '\n\n' : '') + description; const ss = SpreadsheetApp.openById(SHEET_ID); const block = readTerms(ss, 'Hard Block'); const caution = readTerms(ss, 'Caution'); const matches = findMatches(fullText, block, caution); const result = { matches: matches, counts: { block: matches.filter(m => m.tier === 'block').length, caution: matches.filter(m => m.tier === 'caution').length } }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function readTerms(ss, sheetName) { const sheet = ss.getSheetByName(sheetName); if (!sheet) return []; const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; // Column A = term, Column B = reason (optional) const data = sheet.getRange(2, 1, lastRow - 1, 2).getValues(); return data .map(r => ({ term: String(r[0]).trim(), reason: String(r[1] || '').trim() })) .filter(r => r.term); } function findMatches(fullText, blockTerms, cautionTerms) { const allTerms = [ ...blockTerms.map(t => ({ term: t.term, reason: t.reason, tier: 'block' })), ...cautionTerms.map(t => ({ term: t.term, reason: t.reason, tier: 'caution' })) ].sort((a, b) => b.term.length - a.term.length); const lowerText = fullText.toLowerCase(); const matches = []; allTerms.forEach(({ term, reason, tier }) => { const lowerTerm = term.toLowerCase(); if (!lowerTerm) return; let searchFrom = 0; while (true) { const idx = lowerText.indexOf(lowerTerm, searchFrom); if (idx === -1) break; const overlaps = matches.some(m => idx < m.index + m.length && idx + lowerTerm.length > m.index); if (!overlaps) matches.push({ term, reason, tier, index: idx, length: lowerTerm.length }); searchFrom = idx + lowerTerm.length; } }); return matches.sort((a, b) => a.index - b.index); } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: 'ok — use POST' })) .setMimeType(ContentService.MimeType.JSON); }
